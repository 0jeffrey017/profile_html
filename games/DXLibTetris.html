HIGHLIGHT / 01

盤面と操作の実装

盤面の状態をもとに移動と回転を判定し、500ミリ秒ごとに自動落下させます。次のブロックと着地位置を表示して、先を考えて操作できるようにしました。

課題と対応 / Challenges & Solutions

Problem / 問題点

ブロックを回転させたときに、壁や固定ブロックと重ならないようにする必要がありました。

Solution / 解決策

回転中心から各マスの移動先を計算し、すべてのマスが盤面内かつ固定ブロックと重ならない場合にだけ盤面を更新します。

Problem / 問題点

操作中のブロックが、どこに着地するかを表示する必要がありました。

Solution / 解決策

ブロック全体が落下できる距離を調べ、着地先の空きマスに影を配置します。更新前には前フレームの影を消去します。

#include "BlockMovement.h"
#include <DxLib.h>

void BlockMovement::Update(Board& board)
{
    KeyBoard.Update();

    if (KeyBoard.Input_GetKeyboardDown(KEY_INPUT_SPACE))
    {
        TurnClockWise(board);
    }
    else if (KeyBoard.Input_GetKeyboardDown(KEY_INPUT_D))
    {
        Move(board, 1, 0);
    }
    else if (KeyBoard.Input_GetKeyboardDown(KEY_INPUT_A))
    {
        Move(board, -1, 0);
    }
    else if (KeyBoard.Input_GetKeyboardDown(KEY_INPUT_S) || KeyBoard.Input_GetKeyboard(KEY_INPUT_S) >= 60)
    {
        MoveDown(board);
    }
        
}

void BlockMovement::SetPivot(int x, int y, bool isO)
{
    this->pivotX = x;
    this->pivotY = y;
    this->isBlockO = isO;
}

std::array<int, 2> BlockMovement::GetPivot() const
{
    return {pivotX, pivotY};
}

bool BlockMovement::UpdateBlockOnBoardDown(Board& board)
{
    return MoveDown(board);
}

bool BlockMovement::TurnClockWise(Board& board)
{
    if (isBlockO)
    {
        return false;
    }
    const auto before = board.GetTypeInCell(Board::Falling);
    if (before.empty())
    {
        return false;
    }

    std::vector<std::array<int, 2>> after;
    for (const auto& pos : before)
    {
        int x = pivotX - (pos[1] - pivotY);
        int y = pivotY + (pos[0] - pivotX);
        if (!board.IsInside(x, y) || board.ReadAt(x, y) == Board::Fixed)
        {
            return false;
        }
        after.push_back({x, y});
    }

    for (const auto& pos : before)
    {
        board.WriteAt(pos[0], pos[1]) = Board::Empty;
    }
    for (const auto& pos : after)
    {
        board.WriteAt(pos[0], pos[1]) = Board::Falling;
    }
    return true;
}

bool BlockMovement::Move(Board& board, int dx, int dy)
{
    const auto positions = board.GetTypeInCell(Board::Falling);
    if (positions.empty())
    {
        return false;
    }

    for (const auto& pos : positions)
    {
        int x = pos[0] + dx;
        int y = pos[1] + dy;
        if (!board.IsInside(x, y) || board.ReadAt(x, y) == Board::Fixed)
        {
            return false;
        }
    }

    for (const auto& pos : positions)
    {
        board.WriteAt(pos[0], pos[1]) = Board::Empty;
    }
    for (const auto& pos : positions)
    {
        board.WriteAt(pos[0] + dx, pos[1] + dy) = Board::Falling;
    }

    if (!isBlockO){
        pivotX += dx;
        pivotY += dy;
    }
    return true;
}

bool BlockMovement::MoveDown(Board& board)
{
    if (Move(board, 0, 1))
    {
        return true;
    }
    board.FixFallingBlock();
    return false;
}

#include "ShadowSystem.h"

#include <algorithm>

void ShadowSystem::Update(Board& board)
{
    Clear(board);
    auto fallingBlock = board.GetTypeInCell(Board::Falling);
    
    if (fallingBlock.empty())
    {
        return;
    }
    
    int max_y = 0;
    
    for (auto block : fallingBlock)
    {
        max_y = std::max(max_y,block.at(1));
    }

    int dy = 0;
    
    while (true)
    {
        bool canFall = true;

        for (const auto& block : fallingBlock)
        {
            int x = block[0];
            int y = block[1] + dy + 1;

            if (!board.IsInside(x, y) ||
                board.ReadAt(x, y) == Board::Fixed)
            {
                canFall = false;
                break;
            }
        }

        if (!canFall)
        {
            break;
        }

        ++dy;
    }
    for (const auto& block : fallingBlock)
    {
        int x = block[0];
        int y = block[1] + dy;

        if (board.ReadAt(x, y) == Board::Empty)
        {
            board.WriteAt(x, y) = Board::Shadow;
        }
    }
}

void ShadowSystem::Clear(Board& board)
{
    auto fallingBlock = board.GetTypeInCell(Board::Shadow);
    if (fallingBlock.empty())
    {
        return;
    }
    
    for (auto block : fallingBlock)
    {
        board.WriteAt(block.at(0),block.at(1)) = Board::Empty;
    }
}
